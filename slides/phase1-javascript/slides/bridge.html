WEEK 2 · THE BRIDGE

Students already think in Python; JavaScript changes the spelling

PYTHON

def count_open(services):

    total = 0

    for s in services:

        if s["open"]:

            total += 1

    return total

JAVASCRIPT

function countOpen(services) {

  let total = 0;

  for (const s of services) {

    if (s.open) total += 1;

  }

  return total;

}
